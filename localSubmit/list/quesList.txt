// Ques2
func nextPermutation(nums []int)  {
    n:=len(nums)

    bp:= -1
    for i:=n-2;i>=0;i--{
        if nums[i]<nums[i+1] {
            bp=i
            break;
        }
    }
    if bp == -1 {
        for i,j:=0,n-1;i<j;i,j=i+1,j-1 {
            nums[i], nums[j] = nums[j],nums[i]
        }
        return
    }
    for i:=n-1;i>=bp;i--{
        if nums[i]>nums[bp] {
            nums[i], nums[bp] = nums[bp], nums[i]
            break
        }
    }

    for i,j:=bp+1,n-1;i<j;i,j=i+1,j-1 {
        nums[i], nums[j] = nums[j],nums[i]
    }
    
}

func generate(numRows int) [][]int {
    var ans [][]int
    n:=numRows
    // ans:=make([][]int,0)
    for i:=0;i<n;i++ {
        // var temp []int 
        temp:= make([]int,i+1)
        temp[0]=1
        temp[i]=1
        for j:=1;j<i;j++ {
            temp[j]=ans[i-1][j-1]+ans[i-1][j]
        }
        // ans.pushback(temp)
        ans=append(ans,temp)
    }
    return ans 
}

    // kadane's algo
    n:=len(nums)

    sum:=0

        }
    }

    return ans

func maxProfit(prices []int) int {
    //bruteforce ->gives tle
    // // ans:=-int(INF)

    //     }
    // }
    // return ans
    

    //optimise
    n:=len(prices)
    // ans:=-int(INF)
    ans:=0
    minm:=int(INF)
    for i:=0;i<n;i++ {
        if prices[i]<minm {
            minm=prices[i]
        }
        ans=max(prices[i]-minm,ans)
    }
    return ans

}

        }
    }

        for l,h:=0,m-1;l<h;l,h=l+1, h-1{
            matrix[i][l], matrix[i][h]=matrix[i][h],matrix[i][l]
        }
        
    }
    
}

//Ques8
func merge(nums1 []int, m int, nums2 []int, n int)  {
    // single solution inplace
    a,b,t:=m-1,n-1,len(nums1)-1
    for a>=0 && b>=0 {
        if nums1[a]> nums2[b] {
            nums1[t]=nums1[a]
            a=a-1
            t=t-1
        } else {
            nums1[t]=nums2[b]
            b=b-1
            t=t-1
        }

    }

    for b>=0 {
        nums1[t]=nums2[b]
        b=b-1
        t=t-1
    }
}

func ceilDiv(a, b ll) ll {
	if a%b == 0 {
		return a / b
	}
	if a > 0 {
		return a/b + 1
	}
	return a / b
}

func gcd(a, b ll) ll {
	for b != 0 {
		a, b = b, a%b
	}
	if a < 0 {
		return -a
	}
	return a
}

func power(base, p ll) ll {
	ans := ll(1)
	for p > 0 {
		if p%2 == 0 {
			base *= base
			p /= 2
		} else {
			ans *= base
			p--
		}
	}
	return ans
}
