func nextPermutation(nums []int)  {
    n:=len(nums)

    bp:= -1
    for i:=n-2;i>=0;i--{
        if nums[i]<nums[i+1] {
            bp=i
            break;
        }
    }
    if bp == -1 {
        for i,j:=0,n-1;i<j;i,j=i+1,j-1 {
            nums[i], nums[j] = nums[j],nums[i]
        }
        return
    }
    for i:=n-1;i>=bp;i--{
        if nums[i]>nums[bp] {
            nums[i], nums[bp] = nums[bp], nums[i]
            break
        }
    }

    for i,j:=bp+1,n-1;i<j;i,j=i+1,j-1 {
        nums[i], nums[j] = nums[j],nums[i]
    }
    
}

    var ans [][]int

        temp:= make([]int,i+1)
        temp[0]=1
        temp[i]=1
        for j:=1;j<i;j++ {
            temp[j]=ans[i-1][j-1]+ans[i-1][j]
        }
        // ans.pushback(temp)
        ans=append(ans,temp)
    }
    return ans 
}

func ceilDiv(a, b ll) ll {
	if a%b == 0 {
		return a / b
	}
	if a > 0 {
		return a/b + 1
	}
	return a / b
}

func gcd(a, b ll) ll {
	for b != 0 {
		a, b = b, a%b
	}
	if a < 0 {
		return -a
	}
	return a
}

func power(base, p ll) ll {
	ans := ll(1)
	for p > 0 {
		if p%2 == 0 {
			base *= base
			p /= 2
		} else {
			ans *= base
			p--
		}
	}
	return ans
}
