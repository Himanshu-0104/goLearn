//Below are Methods for type(LinkedList)

// insert at end
func (ll *LinkedList) Insert(data int) {
	var newNode *Node
	newNode = &Node{data:data}

	if ll.head==nil {
		ll.head = newNode
		return 
	}

	var temp *Node
	temp = ll.head
	for temp.next != nil {
		temp = temp.next
	}
	temp.next = newNode

}

// delete a node by value //GC handles automatic 
func (ll *LinkedList) Delete(data int) {
	if ll.head == nil {
		return
	}
	if ll.head.data == data {
		ll.head = ll.head.next
		return
	}
	var prev *Node 
	prev = ll.head
	curr:= ll.head.next

	for curr !=nil {
		if curr.data == data {
			prev.next = curr.next
			return
		}
		prev = curr
		curr = curr.next
	}
}


// display list
func (ll *LinkedList) Display() {
	var temp *Node
	temp = ll.head

	for temp != nil {
		fmt.Print(temp.data, " -> ")
		temp=temp.next
	}
	fmt.Println("nil") //last alwasys points to nil

}
